Figure 4.3: Entity Relationship Diagram

Database structure. Transit entities derive from the GTFS feed; users and journey searches support roles, history and reporting.

erDiagram
    STOPS ||--o{ STOP_TIMES : "is visited by"
    STOPS ||--o| STOPS : "parent_station"
    ROUTES ||--o{ TRIPS : "operates"
    TRIPS ||--|{ STOP_TIMES : "consists of"
    USERS ||--o{ JOURNEY_SEARCHES : "performs"
    STOPS ||--o{ JOURNEY_SEARCHES : "is origin of"
    STOPS ||--o{ JOURNEY_SEARCHES : "is destination of"

    STOPS {
        varchar stop_id PK
        varchar stop_name
        double stop_lat
        double stop_lon
        int location_type
        varchar parent_station FK
    }
    ROUTES {
        varchar route_id PK
        varchar agency_id
        varchar route_short_name
        varchar route_long_name
        int route_type
    }
    TRIPS {
        varchar trip_id PK
        varchar route_id FK
        varchar service_id
        varchar trip_headsign
        int direction_id
        varchar shape_id
    }
    STOP_TIMES {
        varchar trip_id PK_FK
        int stop_sequence PK
        varchar stop_id FK
        int arrival_time
        int departure_time
    }
    SHAPES {
        varchar shape_id PK
        int shape_pt_sequence PK
        double shape_pt_lat
        double shape_pt_lon
    }
    USERS {
        bigint id PK
        varchar username UK
        varchar password_hash
        varchar role
        timestamp created_at
    }
    JOURNEY_SEARCHES {
        bigint id PK
        bigint user_id FK
        varchar origin_stop_id FK
        varchar destination_stop_id FK
        boolean route_found
        int transfer_count
        int estimated_minutes
        timestamp searched_at
    }
    VECTOR_STORE {
        uuid id PK
        text content
        jsonb metadata
        vector embedding
    }
